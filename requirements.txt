# cephalo — Python dependencies for utility scripts
#
# Install with:
#   pip install -r requirements.txt
#
# Python 3.9+ required (the scripts use PEP-585 generic type hints).

# The upper bounds are deliberate: a new major release of either library can change
# the APIs these scripts rely on, and an open range would pull it in silently.
# Raise a cap only after testing against the new major.
python-docx>=1.0.0,<2.0    # scripts/abbreviations_extract.py, bibliography_check.py
                           # (1.0+ required for stable doc.part.package.iter_parts())
pypdf[crypto]>=3.0.0,<7.0  # scripts/extract_styleguide.sh (pure-Python PDF extraction);
                           # [crypto] adds the AES support that encrypted, copy-protected
                           # guideline PDFs need (it pulls in the `cryptography` package)

# Optional but recommended for development / contributors:
# pyyaml>=6.0         # if you want to lint/validate the workflow YAMLs yourself
