# cephalo — Python dependencies for utility scripts
#
# Install with:
#   pip install -r requirements.txt
#
# Python 3.9+ required (the scripts use PEP-585 generic type hints).

# The upper bounds are deliberate: a new major release of either library can change
# the APIs these scripts rely on, and an open range would pull it in silently.
# Raise a cap only after testing against the new major.
python-docx>=1.0.0,<2.0    # scripts/abbreviations_extract.py, bibliography_check.py,
                           # verbatim_check.py (Word chapters)
                           # (1.0+ required for stable doc.part.package.iter_parts())
pypdf[crypto]>=6.0.0,<7.0  # scripts/extract_styleguide.sh (pure-Python PDF extraction),
                           # verbatim_check.py (PDF sources);
                           # [crypto] adds the AES support that encrypted, copy-protected
                           # guideline PDFs need (it pulls in the `cryptography` package).
                           # 6.0+ limits how far one compressed stream may expand (later
                           # 6.x releases limit more filters); a crafted PDF can still make
                           # a run slow or memory-hungry, so stop one that hangs. Keep pypdf
                           # current: pip install -U -r requirements.txt. Older versions
                           # would stay installed if present.

# cryptography 49 and later ship macOS wheels for Apple Silicon only. On an
# Intel Mac, pip would pick the newest release and try to build it from source,
# which needs a Rust toolchain and fails; 48.x is the last release with Intel-Mac
# wheels. Other platforms are not affected by this line.
cryptography<49; sys_platform == "darwin" and platform_machine == "x86_64"
